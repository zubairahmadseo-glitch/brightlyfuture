        if (!md || md.trim().length < 200) throw new Error('The draft came back empty. Please try again.');
        return md.trim().replace(/^```(?:markdown|md)?\s*/i, '').replace(/```\s*$/, '');
      });
    }, function (e) { clearTimeout(timer); throw e; });
  }

  function generate() {
    $('genErr').textContent = '';
    var o = { topic: $('topic').value.trim(), kw: $('kw').value.trim(), aud: $('aud').value.trim(), tone: $('tone').value,
      len: $('len').value, sp: $('sp').value, fmt: $('fmt').value, notes: $('notes').value.trim(), faq: $('faq').checked };
    if (o.topic.length < 5) { $('genErr').textContent = 'Please enter a topic (at least a few words).'; $('topic').focus(); return; }
    if (busy || (lastRequest && cooldownLeft() > 0)) return;
    busy = true; lastRequest = Date.now(); setBtn(); progress(true);
    request(o, 0).then(render).catch(function (e) {
      $('genErr').textContent = e.name === 'AbortError' ? 'This took too long. Please try again, or choose a shorter length.' : (e.message === 'Failed to fetch' ? 'Could not reach the AI service. Check your connection and try again.' : e.message);
    }).then(function () { busy = false; lastRequest = Date.now(); progress(false); setBtn(); });
  }

  function init() {
    if (!$('genBtn')) { setTimeout(init, 200); return; }
    $('genBtn').addEventListener('click', function (e) { e.preventDefault(); generate(); });
    $('topic').addEventListener('keydown', function (e) { if (e.key === 'Enter') { e.preventDefault(); generate(); } });
    $('clearBtn').addEventListener('click', function (e) {
      e.preventDefault();
      ['topic', 'kw', 'aud', 'notes'].forEach(function (id) { $(id).value = ''; });
      $('genErr').textContent = ''; $('resultSection').style.display = 'none'; lastMd = ''; setBtn();
    });
    $('sampleBtn').addEventListener('click', function (e) {
      e.preventDefault();
      $('topic').value = 'How to start composting at home in a small garden';
      $('kw').value = 'composting for beginners';
      $('aud').value = 'UK beginners with a small garden or balcony';
      $('fmt').value = 'how-to guide'; $('tone').value = 'Friendly'; $('len').value = '700';
      $('notes').value = 'I use a 220-litre plastic bin from the council. Mix roughly equal amounts of greens (veg peelings, grass) and browns (cardboard, dry leaves). Mine took about 9 months to make usable compost.';
    });
    $('copyMdBtn').addEventListener('click', function (e) { e.preventDefault(); copy(lastMd); });
    $('copyHtmlBtn').addEventListener('click', function (e) { e.preventDefault(); copy(mdToHtml(lastMd).replace(/<\/?mark>/g, '')); });
    $('dlBtn').addEventListener('click', function (e) {
      e.preventDefault(); if (!lastMd) return;
      var a = document.createElement('a'); a.href = URL.createObjectURL(new Blob([lastMd], { type: 'text/markdown' })); a.download = 'blog-draft.md';
      document.body.appendChild(a); a.click(); document.body.removeChild(a);
    });
    $('ctaBtn').addEventListener('click', function (e) { e.preventDefault(); $('topic').scrollIntoView({ behavior: 'smooth', block: 'center' }); $('topic').focus(); });
    var faq = document.querySelectorAll('.faq-q');
    for (var i = 0; i < faq.length; i++) faq[i].addEventListener('click', function (e) { e.preventDefault(); this.parentElement.classList.toggle('open'); });
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init); else init();
})();
</script>
