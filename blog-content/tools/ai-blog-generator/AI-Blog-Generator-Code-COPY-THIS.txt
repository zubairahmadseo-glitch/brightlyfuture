(function () {

  R.innerHTML = '<style>' +

  function $(id) { return R.getElementById(id); }
  var lastMd = '', lastRequest = 0, busy = false;

  var BANNED = ['delve', 'tapestry', 'in today\'s', 'ever-evolving', 'landscape', 'realm', 'unlock', 'unleash', 'harness', 'leverage', 'seamless', 'robust', 'game-changer', 'cutting-edge', 'furthermore', 'moreover', 'additionally', 'in conclusion', 'embark', 'elevate', 'empower', 'foster', 'pivotal', 'crucial', 'navigate'];

  function buildPrompt(o) {
    var sys = 'You are an experienced blog writer. Write in ' + (o.sp === 'UK' ? 'UK English spelling' : 'US English spelling') + '. ' +
      'Output clean Markdown only: one H1 title, then sections with ## headings. No preamble, no closing remarks about the task.';
    var u = 'Write a ' + o.fmt + ' blog post of about ' + o.len + ' words.\n' +
      'Topic: ' + o.topic + '\n' +
      (o.kw ? 'Target keyword: "' + o.kw + '". Use it in the H1, the first paragraph and one or two headings, naturally.\n' : '') +
      (o.aud ? 'Audience: ' + o.aud + '.\n' : '') +
      'Tone: ' + o.tone + ', clear and specific.\n' +
      'Rules:\n' +
      '- Start with a 2–3 sentence introduction that answers the main question directly.\n' +
      '- Write ## headings as the questions a reader would ask. The first sentence under each heading must answer it.\n' +
      '- Short paragraphs (2–4 sentences). Vary sentence length. Use lists only where they help.\n' +
      '- Never invent statistics, studies, quotes, prices or personal stories. Where a fact or real example would help, write a placeholder like [ADD SOURCE: …] or [ADD YOUR EXAMPLE: …].\n' +
      '- Avoid these words and phrases: ' + BANNED.join(', ') + '.\n' +
      (o.faq ? '- End with "## Frequently Asked Questions" containing 3 short Q&As (### for each question) that the body does not already answer.\n' : '- End with a short conclusion section.\n') +
      (o.notes ? '\nUse these facts and notes from the author, and do not contradict them:\n' + o.notes + '\n' : '');
    return [{ role: 'system', content: sys }, { role: 'user', content: u }];
  }

  function esc(s) { return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
  function inline(s) {
    return esc(s).replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>').replace(/\*(.+?)\*/g, '<em>$1</em>')
      .replace(/\[(ADD [^\]]+)\]/g, '<mark>[$1]</mark>');
  }
  function mdToHtml(md) {
    var lines = md.replace(/\r/g, '').split('\n'), html = '', list = null, para = [];
    function flushP() { if (para.length) { html += '<p>' + inline(para.join(' ')) + '</p>'; para = []; } }
    function flushL() { if (list) { html += '</' + list + '>'; list = null; } }
    lines.forEach(function (l) {
      var m;
      if (/^\s*$/.test(l)) { flushP(); flushL(); return; }
      if ((m = l.match(/^(#{1,3})\s+(.*)$/))) { flushP(); flushL(); var n = m[1].length; html += '<h' + n + '>' + inline(m[2]) + '</h' + n + '>'; return; }
      if ((m = l.match(/^\s*[-*]\s+(.*)$/))) { flushP(); if (list !== 'ul') { flushL(); html += '<ul>'; list = 'ul'; } html += '<li>' + inline(m[1]) + '</li>'; return; }
      if ((m = l.match(/^\s*\d+[.)]\s+(.*)$/))) { flushP(); if (list !== 'ol') { flushL(); html += '<ol>'; list = 'ol'; } html += '<li>' + inline(m[1]) + '</li>'; return; }
      flushL(); para.push(l.trim());
    });
    flushP(); flushL();
    return html;
  }

  function syl(w) { w = w.toLowerCase().replace(/[^a-z]/g, ''); if (!w) return 0; if (w.length <= 3) return 1; w = w.replace(/(?:[^laeiouy]es|ed|[^laeiouy]e)$/, '').replace(/^y/, ''); var m = w.match(/[aeiouy]{1,2}/g); return Math.max(1, m ? m.length : 1); }
  function stats(md) {
    var text = md.replace(/^#+\s.*$/gm, '').replace(/[*_\[\]]/g, '');
    var words = text.match(/[A-Za-z0-9][A-Za-z0-9'’-]*/g) || [];
    var sents = (text.match(/[^.!?\n]+[.!?]+/g) || []).length || 1;
    var s = words.reduce(function (a, w) { return a + syl(w); }, 0);
    var fre = Math.round(206.835 - 1.015 * (words.length / sents) - 84.6 * (s / Math.max(1, words.length)));
    var lc = md.toLowerCase(), found = BANNED.filter(function (b) { return lc.indexOf(b) > -1; });
    var ph = (md.match(/\[ADD [^\]]+\]/g) || []).length;
    return { words: words.length, fre: Math.max(0, Math.min(100, fre)), mins: Math.max(1, Math.round(words.length / 238)), banned: found, placeholders: ph };
  }

  function render(md) {
    lastMd = md;
    $('out').innerHTML = mdToHtml(md);
    var s = stats(md);
    $('stats').innerHTML = '<span class="chip">' + s.words + ' words</span><span class="chip">' + s.mins + ' min read</span><span class="chip">Flesch ' + s.fre + '</span>' +
      (s.placeholders ? '<span class="chip" style="background:#FEF3C7">' + s.placeholders + ' places to add your facts</span>' : '');
    $('actions').innerHTML = '<button class="sec" id="cpMd">Copy Markdown</button><button class="sec" id="cpHtml">Copy HTML</button><button class="sec" id="dl">Download .md</button>';
    $('cpMd').onclick = function () { copy(lastMd, this); };
    $('cpHtml').onclick = function () { copy(mdToHtml(lastMd), this); };
    $('dl').onclick = function () { var a = document.createElement('a'); a.href = URL.createObjectURL(new Blob([lastMd], { type: 'text/markdown' })); a.download = 'blog-draft.md'; document.body.appendChild(a); a.click(); a.remove(); };
    var c = $('check'); c.style.display = 'block';
    c.innerHTML = '<b>Before you publish this draft</b>' +
      '1. Replace every highlighted [ADD …] placeholder with your own facts or sources.<br>' +
      '2. Check every claim: AI drafts can state wrong facts confidently.<br>' +
      '3. Add one example, photo or result that only you have.<br>' +
      '4. Aim for Flesch 60–70' + (s.fre < 60 ? ' (this draft is ' + s.fre + ': shorten some sentences)' : '') + '.' +
      (s.banned.length ? '<br>5. Rewrite these AI-sounding words: <i>' + esc(s.banned.join(', ')) + '</i>.' : '') +
      '<br><span class="muted">Next: check it with our <a href="/readability-checker/" target="_blank">Readability Checker</a> and <a href="/ai-blog-detector/" target="_blank">AI Writing Checker</a>.</span>';
  }

  function copy(text, btn) {
    var done = function () { var t = btn.textContent; btn.textContent = 'Copied!'; setTimeout(function () { btn.textContent = t; }, 1500); };
    if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(text).then(done);
    else { var ta = document.createElement('textarea'); ta.value = text; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); ta.remove(); done(); }
  }

  function cooldownLeft() { return Math.max(0, COOLDOWN - Math.floor((Date.now() - lastRequest) / 1000)); }
  function setBtn() {
    var b = $('go'), left = cooldownLeft();
    if (busy) { b.disabled = true; b.innerHTML = '<span class="spin"></span>Writing your draft… (20–60 sec)'; return; }
    if (left > 0) { b.disabled = true; b.textContent = 'Ready again in ' + left + ' s'; setTimeout(setBtn, 1000); return; }
    b.disabled = false; b.textContent = lastMd ? 'Generate another version' : 'Generate draft';
  }

  function generate() {
    $('err').textContent = '';
    var o = {
      topic: $('topic').value.trim(), kw: $('kw').value.trim(), aud: $('aud').value.trim(), tone: $('tone').value,
      len: $('len').value, sp: $('sp').value, fmt: $('fmt').value, notes: $('notes').value.trim(), faq: $('faq').checked
    };
    if (o.topic.length < 5) { $('err').textContent = 'Please enter a topic (at least a few words).'; $('topic').focus(); return; }
    if (busy || cooldownLeft() > 0) return;
    busy = true; lastRequest = Date.now(); setBtn();
    var ctrl = typeof AbortController !== 'undefined' ? new AbortController() : null;
    var timer = setTimeout(function () { if (ctrl) ctrl.abort(); }, 120000);
    fetch(API, {
      method: 'POST', headers: { 'Content-Type': 'application/json' }, signal: ctrl ? ctrl.signal : undefined,
      body: JSON.stringify({ model: 'openai', referrer: REFERRER, private: true, messages: buildPrompt(o) })
    }).then(function (r) {
      if (r.status === 429) throw new Error('The free AI service is busy. Please wait a minute and try again.');
      if (!r.ok) throw new Error('The AI service returned an error (' + r.status + '). Please try again shortly.');
      return r.json();
    }).then(function (d) {
      var md = d && d.choices && d.choices[0] && d.choices[0].message && d.choices[0].message.content;
      if (!md || md.trim().length < 200) throw new Error('The draft came back empty. Please try again.');
      render(md.trim());
    }).catch(function (e) {
      $('err').textContent = e.name === 'AbortError' ? 'This took too long. Please try again, or choose a shorter length.' : e.message;
    }).then(function () { clearTimeout(timer); busy = false; lastRequest = Date.now(); setBtn(); });
  }

  $('go').addEventListener('click', generate);
  $('topic').addEventListener('keydown', function (e) { if (e.key === 'Enter') generate(); });
})();
</script>
