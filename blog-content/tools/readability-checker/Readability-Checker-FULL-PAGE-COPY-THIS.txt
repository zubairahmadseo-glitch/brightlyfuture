      s.hard.forEach(function (h) { body = body.replace(new RegExp('\\b' + h.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '\\b'), '<span class="hl-hard">' + esc(h) + '</span>'); });
      return '<span' + (c ? ' class="' + c.trim() + '"' : '') + ' title="' + esc(tip) + '">' + body + '</span>';
    }).join(' ');

    $('outWords').textContent = r.words; $('outSent').textContent = r.sentences; $('outPara').textContent = r.paras;
    $('outRead').textContent = readTime(r.words); $('outAdv').textContent = r.adv;

    lastReport = 'READABILITY REPORT - BrightlyFuture\n==================================\n' +
      'Flesch Reading Ease: ' + f + ' (' + band(f) + ') | Target: ' + t + '+\n' +
      'Flesch-Kincaid Grade: ' + r.fk.toFixed(1) + '\nGunning Fog: ' + r.fog.toFixed(1) + '\nSMOG: ' + r.smog.toFixed(1) + '\nColeman-Liau: ' + r.cli.toFixed(1) + '\nAverage grade: ' + grade.toFixed(1) + '\n\n' +
      'Words: ' + r.words + ' | Sentences: ' + r.sentences + ' | Paragraphs: ' + r.paras + ' | Reading time: ' + readTime(r.words) + '\n' +
      'Avg words/sentence: ' + r.avg.toFixed(1) + ' | Long sentences: ' + Math.round(r.longPct) + '% | Passive: ' + Math.round(r.pasPct) + '% | Hard words: ' + Math.round(r.hardPct) + '% | -ly adverbs: ' + r.adv + '\n\n' +
      'WHAT TO FIX\n- ' + fixes.map(function (x) { return x.replace(/&ldquo;|&rdquo;/g, '"'); }).join('\n- ') + '\n\nFLAGGED SENTENCES\n' +
      r.sents.map(function (s, i) { var fl = []; if (s.n > 20) fl.push(s.n + ' words'); if (s.passive) fl.push('passive'); if (s.hard.length) fl.push('hard: ' + s.hard.slice(0, 5).join(', ')); return fl.length ? '[' + (i + 1) + '] (' + fl.join('; ') + ') ' + s.text : null; }).filter(Boolean).join('\n') +
      '\n\nhttps://brightlyfuture.co.uk/readability-checker/';
    $('resultSection').style.display = 'block';
  }

  function run() {
    var text = $('inputText').value.trim();
    if (!text) { alert('Please paste some text to check.'); return; }
    if (wordsOf(text).length < 10) { alert('Please paste at least 10 words.'); return; }
    render(analyse(text, +$('target').value));
  }
  function toast() { var t = $('toast'); if (!t) return; t.classList.add('show'); setTimeout(function () { t.classList.remove('show'); }, 2000); }

    $('inputText').addEventListener('input', function () { $('wordCount').textContent = wordsOf(this.value).length; $('charCount').textContent = this.value.length; });
    $('checkBtn').addEventListener('click', function (e) { e.preventDefault(); run(); });
    $('target').addEventListener('change', function () { if ($('resultSection').style.display === 'block') run(); });
    $('clearBtn').addEventListener('click', function (e) { e.preventDefault(); $('inputText').value = ''; $('inputText').dispatchEvent(new Event('input')); $('resultSection').style.display = 'none'; });
    $('sampleBtn').addEventListener('click', function (e) {
      e.preventDefault();
      $('inputText').value = 'Readability is a measure of how easily a reader can understand a piece of written text, and it is determined by a combination of factors including sentence length, word complexity and the overall organisation of the content.\n\nShort sentences help. So do everyday words.\n\nIt has been demonstrated by numerous studies that content which is written in a clear and straightforward manner is significantly more likely to be read in its entirety, shared with others and remembered afterwards. Dr. Smith, for example, found that readers skim most web pages. Keep paragraphs short and use headings so people can find what they need.';
      $('inputText').dispatchEvent(new Event('input'));
    });

      e.preventDefault();
      if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(lastReport).then(toast);

  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init); else init();
})();
